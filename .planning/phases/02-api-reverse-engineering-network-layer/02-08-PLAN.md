## Error model

#   3. ./tools/manual-smoke.sh --dry   # parse env, print sanitized URLs, do NO requests

# EXIT CODES:
#   0 — login + grades fetch returned non-empty JSON shape
#   1 — missing creds / .env.local missing / curl error
#   2 — login HTTP non-2xx
#   3 — grades HTTP non-2xx

fi

REPO_ROOT="$(cd "$SCRIPT_DIR/.." && pwd)"

HOST="https://journal.school28-kirov.ru"

  echo "First 500 bytes:"

rm -f /tmp/manual-smoke-login.body /tmp/manual-smoke-grades.body
